% 3 -- Problem Statement & Motivation
\begin{frame}{Problem Statement \& Motivation}
\small
\begin{block}{Problem Statement}
Municipal waste routes are typically \textbf{static and fixed}, ignoring the real shortest route between homes, bins, and the depot -- wasting fuel, time, and labor. Hazardous waste is handled identically to ordinary waste despite needing immediate, restricted-zone pickup, and fixed schedules over- or under-service locations with different fill rates.
\end{block}
\vspace{1mm}
\begin{columns}[T,totalwidth=\textwidth]
\column{.48\textwidth}
\textbf{\color{GEUBlue}Motivation}
\begin{itemize}\itemsep1pt
\item Fixed routes ignore the real shortest path in the city's road/graph network
\item Hazardous waste in normal queues risks delayed, unsafe transit
\item Calendar-based schedules waste labor on locations nowhere near full
\end{itemize}
\column{.48\textwidth}
\textbf{\color{GEUBlue}Target Users / Stakeholders}
\begin{itemize}\itemsep1pt
\item Municipal waste collection departments and truck operators
\item Residents and commercial establishments being served
\item Public health -- faster, safer hazardous waste clearance
\end{itemize}
\end{columns}
\end{frame}

% 4 -- Objectives
\begin{frame}{Objectives}
\textbf{\color{GEUBlue}Project Objectives}
\vspace{1mm}
\begin{enumerate}\itemsep4pt
\item Model the city as a weighted graph -- depot, homes, junctions, and bins -- with every node and edge tagged by zone type (residential, commercial, industrial).
\item Compute optimal collection routes using Dijkstra's shortest-path algorithm, with two selectable strategies: Priority mode and FIFO mode.
\item Detect hazardous waste and dispatch it immediately via a zone-constrained shortest path that avoids residential and commercial areas.
\item Dynamically determine collection frequency per location from its fill-rate, instead of a fixed calendar.
\item Expose all functionality through a REST API and visualize live, animated route computation on a web frontend.
\end{enumerate}
\end{frame}

% 5 -- Proposed Solution: Approach
\begin{frame}{Proposed Solution -- Approach}
\small
\textbf{\color{GEUBlue}Proposed Approach}
\vspace{1mm}
\begin{enumerate}\itemsep3pt
\item \textbf{City layout} -- depot, homes, junctions, bins, and zone tags loaded from a JSON configuration file at server startup.
\item \textbf{Routing engine} -- Dijkstra's algorithm (plain and zone-constrained variants) computes shortest paths between any two nodes.
\item \textbf{Route builders} -- Priority and FIFO strategies sequence multi-stop collection routes.
\item \textbf{Scheduler} -- fill-rate and last-collected timestamps determine which locations are due for a visit.
\item \textbf{Delivery} -- results served as JSON over a REST API, animated live in a React frontend using Cytoscape.js.
\end{enumerate}
\vfill
\smallnote{A single persistent C++ HTTP server owns the graph and performs routing, classification, and scheduling.}
\end{frame}

% 13 -- Dijkstra's Algorithm: How It Works
\begin{frame}{Dijkstra's Algorithm -- How It Works}
\footnotesize
\begin{center}
\begin{tikzpicture}[
node distance=5mm,
box/.style={rectangle,rounded corners,draw=GEUBlue,fill=GEULightBlue,minimum width=1.95cm,minimum height=9mm,align=center,font=\tiny},
arrow/.style={-{Latex[length=2mm]},thick,draw=GEUBlue}
]
\node[box] (a) {Init \texttt{dist[]},\\push source};
\node[box,right=of a] (b) {Pop min-\\dist node $u$};
\node[box,right=of b] (c) {Relax\\edges of $u$};
\node[box,right=of c] (d) {Reconstruct\\path};
\draw[arrow] (a)--(b);
\draw[arrow] (b)--(c);
\draw[arrow] (c)--(d);
\draw[arrow] (c.north) to[out=90,in=90] node[midway,above,font=\tiny]{repeat} (b.north);
\end{tikzpicture}
\end{center}
\vspace{1mm}
\begin{enumerate}\itemsep2pt
\item \texttt{dist[source]=0}, all others $\infty$; push source into a min-heap keyed by distance.
\item Pop node \texttt{u} with smallest tentative distance.
\item \textbf{Relax} edge \texttt{(u,v,w)}: if \texttt{dist[u]+w < dist[v]}, update \texttt{dist[v]} and push \texttt{v}.
\item Reconstruct the path by walking a predecessor map back to the source.
\end{enumerate}
\vfill
\smallnote{Complexity: $O((V+E)\log V)$ with a binary heap.}
\end{frame}

% 15 -- Fill-Rate & Frequency Management: Formula
\begin{frame}[fragile]{Fill-Rate \& Frequency Management -- Interval Formula}
\footnotesize
Each home/bin stores \texttt{fillRate} (units/day), \texttt{capacity}, and \texttt{lastCollected}. The scheduler turns these into a per-location visit interval:
\vspace{1mm}
\begin{center}
\colorbox{GEULightBlue}{\parbox{.7\textwidth}{\centering
\texttt{interval\_days = capacity / fillRate}, clamped to $[1, 14]$ days}}
\end{center}
\vspace{1mm}
\begin{center}
\begin{tikzpicture}[font=\scriptsize]
\draw[thick,GEUBlue,-{Latex[length=2mm]}] (0,0) -- (10.5,0);
\draw[thick,GEUBlue] (0,0.15) -- (0,-0.15) node[below,yshift=-1mm]{lastCollected};
\draw[thick,GEUBlue] (5,0.15) -- (5,-0.15) node[below,yshift=-1mm]{due (+interval)};
\draw[thick,red] (8,0.15) -- (8,-0.15) node[below,yshift=-1mm]{now};
\draw[decorate,decoration={brace,amplitude=4pt},thick,red] (5,0.4) -- (8,0.4) node[midway,above,yshift=2mm]{overdue-ness $\to$ max-heap rank};
\node[above,font=\tiny] at (2.5,0) {interval\_days};
\end{tikzpicture}
\end{center}
\begin{itemize}\itemsep1pt
\item \textbf{Floor of 1 day} -- stops an extremely fast-filling bin from demanding more than one visit per day
\item \textbf{Ceiling of 14 days} -- guarantees no location goes unvisited more than two weeks, however slowly it fills
\item Due when \texttt{now} $\ge$ \texttt{lastCollected + interval\_days}; \texttt{markCollected()} resets \texttt{lastCollected = now} once visited
\end{itemize}
\end{frame}

% 17 -- Waste Classification Subsystem
\begin{frame}{Waste Classification Subsystem}
\small
\begin{itemize}\itemsep4pt
\item Keyword rules loaded from \texttt{waste\_rules.json}: each category maps to a keyword list and a hazard flag.
\item Categories: recyclable, compost, general, \textbf{medical}, \textbf{chemical}.
\item A description is matched against \textbf{medical} and \textbf{chemical} keywords \textit{first} -- hazard classification always takes priority, even if a recyclable keyword also appears (e.g. ``plastic syringe'' $\to$ medical).
\item Falls back to \texttt{general} if no keyword matches.
\item Output: bin type + a boolean hazardous flag, which triggers the constrained-routing path instead of the normal queue/schedule.
\end{itemize}
\end{frame}

% 18 -- REST API: How It Works & Design
\begin{frame}{REST API -- How It Works \& Design}
\small
The frontend and backend communicate over plain HTTP: the browser sends a JSON request to a resource path, the C++ server responds with a JSON body -- no session state is kept on either side between calls.
\vspace{2mm}
\begin{center}
\renewcommand{\arraystretch}{1.25}
\begin{tabularx}{.96\textwidth}{@{}>{\ttfamily}p{3.1cm}X@{}}
\toprule
Endpoint & Purpose\\
\midrule
GET /graph & Full node/edge list as JSON, including zone tags\\
POST /route & Home list + mode (priority/fifo) $\to$ path \& cost\\
POST /classify & Waste description $\to$ bin type \& hazard flag\\
POST /route/hazard & Hazardous location $\to$ zone-constrained route\\
GET /schedule & Homes/bins due for collection today\\
\bottomrule
\end{tabularx}
\end{center}
\vfill
\smallnote{Uniform error format \texttt{\{ok:false, error, message\}} with HTTP 400/404/422/500 as appropriate.}
\end{frame}

% 19 -- Frontend Visualization
\begin{frame}{Frontend Visualization -- React + Cytoscape.js}
\small
\begin{itemize}\itemsep3pt
\item \textbf{React} -- component-based UI (controls, mode toggle, waste-input form, schedule sidebar) with state managed via hooks; built with Vite into a static bundle.
\item \textbf{Cytoscape.js} -- wrapped inside a dedicated React component for the graph canvas; a graph-drawing library purpose-built for nodes-and-edges visualization.
\item \textbf{Render pipeline:} \texttt{GET /graph} $\to$ build Cytoscape elements $\to$ style by node type/zone $\to$ auto-layout, all inside a \texttt{useEffect}.
\item \textbf{Route animation:} walk each returned path segment, sequentially highlighting edges and pulsing nodes; hazardous routes render in a distinct dashed/red style.
\item \textbf{Live schedule:} periodic \texttt{GET /schedule} polling flags due locations on the map without a page reload.
\end{itemize}
\end{frame}
